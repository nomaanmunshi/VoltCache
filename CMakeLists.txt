cmake_minimum_required(VERSION 3.20)
project(VoltCache VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VOLTCACHE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

find_package(Threads REQUIRED)
add_library(voltcache_core src/cache.cpp src/protocol.cpp src/commands.cpp)
target_include_directories(voltcache_core PUBLIC include)
add_library(asio INTERFACE)
target_include_directories(asio SYSTEM INTERFACE third_party/asio/include)
target_compile_definitions(asio INTERFACE ASIO_STANDALONE)
target_link_libraries(asio INTERFACE Threads::Threads)

foreach(name server cli)
    add_executable(voltcache-${name} src/${name}.cpp)
    target_link_libraries(voltcache-${name} PRIVATE voltcache_core asio)
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(voltcache-unit-tests tests/unit.cpp)
    target_link_libraries(voltcache-unit-tests PRIVATE voltcache_core)
    add_test(NAME unit COMMAND voltcache-unit-tests)
    find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
             --server $<TARGET_FILE:voltcache-server> --cli $<TARGET_FILE:voltcache-cli>)
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
endif()

foreach(target voltcache_core voltcache-server voltcache-cli voltcache-unit-tests)
    if(TARGET ${target})
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
            if(VOLTCACHE_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()

install(TARGETS voltcache-server voltcache-cli RUNTIME DESTINATION bin)
